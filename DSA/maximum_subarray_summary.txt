public class BruteForceMaximumSubarray {

    public static int maxSubArray(int[] nums) {

        int maxSum = Integer.MIN_VALUE;

        // Pick every starting index
        for (int start = 0; start < nums.length; start++) {

            int currentSum = 0;

            // Extend subarray one step at a time
            for (int end = start; end < nums.length; end++) {

                currentSum += nums[end];

                // Update global maximum
                if (currentSum > maxSum) {
                    maxSum = currentSum;
                }
            }
        }

        return maxSum;
    }

    public static void main(String[] args) {

        int[] nums = {-2, 1, -3, 4, -1, 2, 1, -5, 4};

        int result = maxSubArray(nums);

        System.out.println("Maximum Subarray Sum: " + result);
    }
}

public class PrefixSumMaximumSubarray {

    public static int maxSubArray(int[] nums) {

        int n = nums.length;

        // Prefix array stores cumulative sums
        int[] prefix = new int[n];

        prefix[0] = nums[0];

        // Build prefix sums
        for (int i = 1; i < n; i++) {
            prefix[i] = prefix[i - 1] + nums[i];
        }

        int maxSum = Integer.MIN_VALUE;

        // Try every possible subarray
        for (int start = 0; start < n; start++) {

            for (int end = start; end < n; end++) {

                int currentSum;

                // Subarray starts from index 0
                if (start == 0) {
                    currentSum = prefix[end];
                }

                // Remove previous prefix contribution
                else {
                    currentSum = prefix[end] - prefix[start - 1];
                }

                // Update maximum
                if (currentSum > maxSum) {
                    maxSum = currentSum;
                }
            }
        }

        return maxSum;
    }

    public static void main(String[] args) {

        int[] nums = {-2, 1, -3, 4, -1, 2, 1, -5, 4};

        int result = maxSubArray(nums);

        System.out.println("Maximum Subarray Sum: " + result);
    }
}

public class KadaneMaximumSubarray {

    public static int maxSubArray(int[] nums) {

        int maxSum = Integer.MIN_VALUE;

        // Running subarray sum
        int currentSum = 0;

        // Traverse array once
        for (int num : nums) {

            currentSum += num;

            // Update best answer
            if (currentSum > maxSum) {
                maxSum = currentSum;
            }

            // Negative sums hurt future subarrays
            if (currentSum < 0) {
                currentSum = 0;
            }
        }

        return maxSum;
    }

    public static void main(String[] args) {

        int[] nums = {-2, 1, -3, 4, -1, 2, 1, -5, 4};

        int result = maxSubArray(nums);

        System.out.println("Maximum Subarray Sum: " + result);
    }
}
