# ============================================================
# 1. BUBBLE SORT
# ============================================================
# HOW IT WORKS:
#   Walk through the array comparing adjacent pairs. If a pair
#   is out of order, swap them. After one full pass, the largest
#   element "bubbles" to the end. Repeat, shrinking the range by
#   1 each pass since the tail is already sorted.
#
#   The `swapped` flag makes it ADAPTIVE: if a pass has no swaps,
#   the array is already sorted → best case O(n).
#
# TRACE: [5, 3, 8, 1]
#   Pass 1: [3,5,8,1] → [3,5,8,1] → [3,5,1,8]   ← 8 bubbled to end
#   Pass 2: [3,5,1,8] → [3,1,5,8]                ← 5 bubbled
#   Pass 3: [1,3,5,8]                            ← 3 bubbled
#
# COMPLEXITY:
#   Best: O(n) | Average: O(n²) | Worst: O(n²)
#   Space: O(1) | Stable: Yes
# ============================================================
def bubble_sort(arr):
    n = len(arr)
    for i in range(n - 1):
        for j in range(n - 1 - i):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
    return arr


# ============================================================
# 2. SELECTION SORT
# ============================================================
# HOW IT WORKS:
#   Split the array into a SORTED PREFIX (left) and an UNSORTED
#   SUFFIX (right). Each pass, find the minimum in the unsorted
#   part and swap it with the first unsorted element.
#
#   Always does O(n²) comparisons, but at most n−1 swaps → good
#   when write operations are expensive.
#
# TRACE: [64, 25, 12, 22, 11]
#   Pass 1: swap 64 ↔ 11 → [11, 25, 12, 22, 64]
#   Pass 2: swap 25 ↔ 12 → [11, 12, 25, 22, 64]
#   Pass 3: swap 25 ↔ 22 → [11, 12, 22, 25, 64]
#   Pass 4: no swap       → [11, 12, 22, 25, 64]
#
# COMPLEXITY:
#   Best: O(n²) | Average: O(n²) | Worst: O(n²)
#   Space: O(1) | Stable: No
# ============================================================
def selection_sort(arr):
    n = len(arr)
    for i in range(n - 1):
        min_idx = i
        for j in range(i + 1, n):     # find the minimum in unsorted part
            if arr[j] < arr[min_idx]:
                min_idx = j
        arr[i], arr[min_idx] = arr[min_idx], arr[i]   # swap into place
    return arr


# ============================================================
# 3. INSERTION SORT
# ============================================================
# HOW IT WORKS:
#   Like sorting a hand of playing cards: take the next card and
#   INSERT it into its correct spot among the already-sorted cards.
#   Elements larger than `key` shift right to make room.
#
#   ADAPTIVE: nearly-sorted input → almost O(n) because the inner
#   loop barely runs.
#
# TRACE: [5, 2, 4, 6, 1]
#   i=1: key=2 → [2, 5, 4, 6, 1]
#   i=2: key=4 → [2, 4, 5, 6, 1]
#   i=3: key=6 → [2, 4, 5, 6, 1]  (already in place)
#   i=4: key=1 → [1, 2, 4, 5, 6]
#
# COMPLEXITY:
#   Best: O(n) | Average: O(n²) | Worst: O(n²)
#   Space: O(1) | Stable: Yes
# ============================================================
def insertion_sort(arr):
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        # Shift larger elements one position right
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key
    return arr


# ============================================================
# 4. MERGE SORT
# ============================================================
# HOW IT WORKS:
#   DIVIDE & CONQUER: split the array in half, recursively sort
#   each half, then MERGE two sorted halves into one.
#   Merging compares the front of each half and takes the smaller
#   — this is the key O(n) step.
#
#   Always O(n log n) — no worst-case surprise. But needs O(n)
#   extra space.
#
# TRACE: [38, 27, 43, 3]
#   Split:   [38,27]        [43,3]
#   Split:  [38] [27]      [43] [3]
#   Merge:  [27,38]        [3,43]
#   Merge:  [3,27,38,43]
#
# COMPLEXITY:
#   Best: O(n log n) | Average: O(n log n) | Worst: O(n log n)
#   Space: O(n) | Stable: Yes
# ============================================================
def merge_sort(arr):
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    return merge(left, right)


def merge(left, right):
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:      # <= keeps it stable
            result.append(left[i]); i += 1
        else:
            result.append(right[j]); j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result


# ============================================================
# 5. QUICK SORT
# ============================================================
# HOW IT WORKS:
#   Pick a PIVOT and PARTITION the array so that everything smaller
#   is left of the pivot and everything larger is right. The pivot
#   is now in its final position → recursively sort the two sides.
#
#   FASTEST IN PRACTICE (great cache behavior, low constant factor),
#   but worst case is O(n²) when the pivot is always the min/max
#   (e.g., already-sorted input with last-element pivot).
#
#   TIP: Use random or median-of-three pivot to avoid O(n²) on
#   sorted inputs.
#
# TRACE: [10, 7, 8, 9, 1, 5], pivot=5
#   Partition: [1, 5, 8, 9, 10, 7]  → pivot 5 at index 1
#   Recurse left:  [1]
#   Recurse right: [8, 9, 10, 7] → partition around 7 → [7, 8, 9, 10]
#   Final: [1, 5, 7, 8, 9, 10]
#
# COMPLEXITY:
#   Best: O(n log n) | Average: O(n log n) | Worst: O(n²)
#   Space: O(log n) | Stable: No
# ============================================================
def quick_sort(arr, low=0, high=None):
    if high is None:
        high = len(arr) - 1
    if low < high:
        p = partition(arr, low, high)
        quick_sort(arr, low, p - 1)
        quick_sort(arr, p + 1, high)
    return arr


def partition(arr, low, high):
    pivot = arr[high]          # Lomuto partition, pivot = last element
    i = low - 1
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1


# ============================================================
# 6. HEAP SORT
# ============================================================
# HOW IT WORKS:
#   Treat the array as a BINARY HEAP (a complete binary tree
#   stored in an array).
#
#   Phase 1 — Build max-heap: every parent is ≥ its children,
#             so the largest element sits at index 0.
#   Phase 2 — Extract: repeatedly swap the root (max) with the
#             last unsorted element, shrink the heap, re-heapify.
#
#   O(n log n) GUARANTEED and IN-PLACE (O(1) space), but not
#   stable and slower than quicksort in practice due to poor
#   cache locality.
#
# TRACE: [4, 10, 3, 5, 1]
#   Build max-heap: [10, 5, 3, 4, 1]
#   Swap root↔last: [1, 5, 3, 4 | 10] → heapify → [5, 4, 3, 1 | 10]
#   Swap root↔last: [1, 4, 3 | 5, 10] → heapify → [4, 1, 3 | 5, 10]
#   ... Final: [1, 3, 4, 5, 10]
#
# COMPLEXITY:
#   Best: O(n log n) | Average: O(n log n) | Worst: O(n log n)
#   Space: O(1) | Stable: No
# ============================================================
def heap_sort(arr):
    n = len(arr)
    # Build a max-heap
    for i in range(n // 2 - 1, -1, -1):
        heapify(arr, n, i)
    # Extract elements one by one
    for i in range(n - 1, 0, -1):
        arr[0], arr[i] = arr[i], arr[0]   # move max to the end
        heapify(arr, i, 0)
    return arr


def heapify(arr, n, i):
    largest = i
    left, right = 2 * i + 1, 2 * i + 2
    if left < n and arr[left] > arr[largest]:
        largest = left
    if right < n and arr[right] > arr[largest]:
        largest = right
    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]
        heapify(arr, n, largest)



# WHICH ONE TO USE?
#   - Tiny/nearly-sorted arrays        → Insertion Sort
#   - Need guaranteed O(n log n)+stable → Merge Sort
#   - Fastest general purpose          → Quick Sort (randomized pivot)
#   - In-place + guaranteed O(n log n) → Heap Sort
#   - Teaching / simplicity            → Bubble or Selection Sort
# ============================================================
